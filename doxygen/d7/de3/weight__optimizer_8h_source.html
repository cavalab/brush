44 {
45 return (*this)(parameters.data(), residuals.data());
46 }

48 template <typename T>
49 auto operator()(T const* parameters, T* residuals) const -> bool
50 {
51 const T ** new_weights = &parameters;
52 auto residualMap = Eigen::Map<Array<T, Dynamic, 1>>(
53 residuals, GetDataset().get_n_samples());
54
55 if constexpr (PT::program_type == ProgramType::MulticlassClassifier)
56 {
57 using MatrixType = Array<T, Dynamic, Dynamic>;
58 auto probabilities = GetProgram().template predict_with_weights<MatrixType>(
59 GetDataset(), new_weights);
60 for (int i = 0; i < GetDataset().get_n_samples(); ++i)
61 {
62 const int label = static_cast<int>(GetTarget()(i));
63 const T probability = probabilities(i, label);
64 const float weight = class_weights_.empty() ? 1.0f
65 : class_weights_.at(label);
66 // TinySolver minimizes squared residuals, so sqrt(loss)
67 // makes its objective the weighted multinomial log loss.
68 residualMap(i) = sqrt(T(weight) * -log(probability));
69 }
70 }
71 else
72 {
73 using ArrayType = Array<T, Dynamic, 1>;
74 ArrayType y_pred = GetProgram().template predict_with_weights<ArrayType>(
75 GetDataset(), new_weights);
76
77 if (GetDataset().classification)
78 {
79 // tolerance to avoid numeric errors.
80
81 // Using an eps with 7 significant digits to avoid weird behavior.
82 // From some experiments, I noticed that using very small values (previous
83 // was pow(10,-10) caused the optimizer to evaluate to nans), probably because
84 // we were calculating log(0) due to cpp ignoring more than 7 significant digits
85 // for floats. using -6. As a reference, tensorflow uses -7.
86 // We need to remember that we dont want to have eps too big, otherwise
87 // we are biasing the predictions away from 0 and 1
88 float eps = 1e-6f;
89
90 auto y = GetTarget();
91
92 // cout << T(1.0) << ", " << T(eps) << endl;
93
94 // clamp values and avoid log(0)
95 y_pred = y_pred.min(T(1.0) - T(eps)).max(T(eps));
96
97 for (int i = 0; i < y_pred.size(); ++i)
98 {
99 const float weight = class_weights_.empty() ? 1.0f
100 : class_weights_.at(static_cast<int>(y(i)));
101 const T log_loss = -(T(y(i)) * log(y_pred(i))
102 + (T(1.0f) - T(y(i))) * log(T(1.0f) - y_pred(i)));
103 // See multiclass branch above: this makes the least-squares
104 // objective equal weighted binary log loss.
105 residualMap(i) = sqrt(T(weight) * log_loss);
106 }
107 }
108 else
109 {
110 residualMap = y_pred - GetTarget();
111 }
112 }
113
114 return true;
115 }
