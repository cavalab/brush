71State check_type(const ArrayXf& x, const string t)
72{
73 State tmp;
74
75 if (!t.empty())
76 {
77 // Use DataNameType to get the statetype given the string representation
78 DataType feature_type = DataNameType.at(t);
79
80 if (feature_type == DataType::ArrayB)
81 tmp = ArrayXb(x.cast<bool>());
82 else if (feature_type == DataType::ArrayI)
83 tmp = ArrayXi(x.cast<int>());
84 else if (feature_type == DataType::ArrayF)
85 tmp = ArrayXf(x.cast<float>());
86 else
87 HANDLE_ERROR_THROW(
88 "Invalid feature type. check_type does not support this type: " + t);
89 }
90 else
91 {
92 // get feature types (binary or continuous/categorical)
93 bool isBinary = true;
94 bool isCategorical = true;
95
96 std::map<float, bool> uniqueMap;
97 for(int i = 0; i < x.size(); i++)
98 {
99 if(x(i) != 0 && x(i) != 1)
100 isBinary = false;
101 if(x(i) != floor(x(i)) && x(i) != ceil(x(i)))
102 isCategorical = false;
103 else
104 uniqueMap[x(i)] = true;
105 }
106
107 if (isBinary)
108 {
109 tmp = ArrayXb(x.cast<bool>());
110 }
111 else
112 {
113 if(isCategorical && uniqueMap.size() <= 10)
114 {
115 tmp = ArrayXi(x.cast<int>());
116 }
117 else
118 {
119 tmp = x;
120 }
121 }
122 }
123
124 return tmp;
125}

134
135 return x;

139Dataset Dataset::operator()(const vector<size_t>& idx) const
140{
141 std::map<std::string, State> new_features;
142 for (auto& [key, value] : this->features)
143 {
144 auto& k = key;
145 std::visit([&](auto&& arg)
146 {
147 using T = std::decay_t<decltype(arg)>;
148 if constexpr ( T::NumDimensions == 1)
149 new_features[k] = T(arg(idx));
150 else if constexpr (T::NumDimensions==2)
151 new_features[k] = T(arg(idx, Eigen::internal::all_t{}));
152 else
153 static_assert(always_false_v<T>, "non-exhaustive visitor!");
154 },
155 value
156 );
157 }
158 ArrayXf new_y;
159 if (this->y.size()>0)
160 {
161 new_y = this->y(idx);
162 }
163 // using constructor 1
164 Dataset result(new_features, new_y, this->classification);
165 // Preserve the original feature name and type order from parent
166 // The constructor's init() may have reordered them, so we fix that here
167 result.feature_names = this->feature_names;
168 result.feature_types = this->feature_types;
169 result.feature_name_order_ = this->feature_names;
170
171 // Rebuild features_of_type to match the corrected order
172 result.features_of_type.clear();
173 result.unique_data_types.clear();
174 for (size_t i = 0; i < result.feature_names.size(); ++i) {
175 const auto& name = result.feature_names[i];
176 const auto& ftype = result.feature_types[i];
177 result.features_of_type[ftype].push_back(name);
178 Util::unique_insert(result.unique_data_types, ftype);
179 }
180
181 return result;
182}

185Dataset Dataset::get_batch() const
186{
187 if (!use_batch)
188 return (*this);
189
190 const auto total_samples = static_cast<size_t>(this->get_n_samples());
191 const auto target_batch_size = static_cast<size_t>(ceil(total_samples * batch_size));
192
193 // Start a new epoch only after every row from the preceding epoch has
194 // been returned. This prevents replacement before a complete pass through
195 // the training partition.
196 if (batch_cycle_idx.empty() || batch_cycle_batch_cursor >= batch_cycle_sizes.size())
197 {
198 // resets everything
199 batch_cycle_idx.clear();
200 batch_cycle_sizes.clear();
201 batch_cycle_cursor = 0;
202 batch_cycle_batch_cursor = 0;
203
204 // repopulate the indexes
205 if (classification)
206 {
207 // Shuffle each class queue. Every batch preserves the original
208 // class prevalence and reserves one slot per class. Once a class
209 // queue is exhausted, that class is sampled with replacement so
210 // later batches still train against it while larger class queues
211 // complete their no-replacement pass.
212 std::map<float, vector<size_t>> class_indices;
213 for (size_t i = 0; i < total_samples; ++i)
214 class_indices[y(i)].push_back(i);
215
216 for (auto& [label, indices] : class_indices)
217 r.shuffle(indices.begin(), indices.end());
218
219 // creating the batches
220 vector<size_t> positions(class_indices.size(), 0); // this gets incremented everytime a row is added to a batch
221 const auto class_count = class_indices.size();
222 const auto current_batch_size = std::max(target_batch_size, class_count);
223 while (true) // this while(true) iterates once for each batch creation
224 {
225 // searches for classes with unseen samples
226 bool has_unseen_rows = false;
227 size_t class_position = 0;
228 for (const auto& [label, indices] : class_indices) {
229 has_unseen_rows |= positions[class_position] < indices.size();
230 ++class_position;
231 }
232 if (!has_unseen_rows) // all rows were added to a batch -- this is the escape condition to the while (true) loop
233 break;
234
235 vector<size_t> counts(class_indices.size(), 0); // will count how many samples per class for the current batch
236 vector<std::pair<float, size_t>> remainders; // will store how many samples are left per class after the current batch
237
238 // free space in the batch. Notice that one sample per class is already being considered here
239 const auto allocatable_slots = current_batch_size - class_count;
240
241 class_position = 0;
242 for (const auto& [label, indices] : class_indices)
243 {
244 counts[class_position] = 1; // this ensures one sample per class
245
246 // find rounded number of slots that will be used for the class
247 // to ensure they keep approximate prevalence w.r.t. original data
248 const float exact = static_cast<float>(indices.size() * allocatable_slots)
249 / static_cast<float>(total_samples);
250 const auto floor_allocation = static_cast<size_t>(std::floor(exact));
251
252 counts[class_position] += floor_allocation;
253 remainders.emplace_back(exact - std::floor(exact), class_position);
254 ++class_position;
255 }
256
257 std::stable_sort(remainders.begin(), remainders.end(),
258 [](const auto& left, const auto& right) {
259 return left.first > right.first;
260 });
261
262 size_t assigned = class_count;
263 for (const auto count : counts)
264 assigned += count - 1;
265 for (size_t i = 0; assigned < current_batch_size; ++i, ++assigned)
266 ++counts[remainders.at(i).second];
267
268 // filling the batch
269 vector<size_t> batch_indices;
270 batch_indices.reserve(current_batch_size);
271 class_position = 0;
272 for (const auto& [label, indices] : class_indices)
273 {
274 const auto available = indices.size() - positions[class_position];
275 const auto unique_count = std::min(counts[class_position], available);
276 for (size_t i = 0; i < unique_count; ++i)
277 batch_indices.push_back(indices[positions[class_position]++]);
278
279 // Repeat only a class that was already exhausted before
280 // this batch began. If its final unseen row was consumed
281 // above, return a smaller final batch instead of repeating
282 // that row immediately.
283 if (available == 0) {
284 for (size_t i = 0; i < counts[class_position]; ++i)
285 batch_indices.push_back(indices.at(r.rnd_int(0, indices.size() - 1)));
286 }
287 ++class_position;
288 }
289 r.shuffle(batch_indices.begin(), batch_indices.end());
290 batch_cycle_idx.insert(batch_cycle_idx.end(), batch_indices.begin(), batch_indices.end());
291 batch_cycle_sizes.push_back(batch_indices.size());
292 }
293 }
294 else
295 {
296 // Regression batches are stratified by target rank. The four
297 // equal-frequency y quartiles are shuffled independently, then
298 // apportioned evenly into each batch without replacement.
299 vector<size_t> sorted_indices(total_samples);
300 std::iota(sorted_indices.begin(), sorted_indices.end(), 0);
301 std::stable_sort(sorted_indices.begin(), sorted_indices.end(),
302 [this](size_t left, size_t right) { return y(left) < y(right); });
303
304 vector<vector<size_t>> quartiles(4);
305 for (size_t rank = 0; rank < total_samples; ++rank)
306 quartiles.at((rank * quartiles.size()) / total_samples).push_back(sorted_indices.at(rank));
307
308 size_t nonempty_quartiles = 0;
309 for (auto& quartile : quartiles) {
310 if (!quartile.empty()) {
311 r.shuffle(quartile.begin(), quartile.end());
312 ++nonempty_quartiles;
313 }
314 }
315
316 vector<size_t> positions(quartiles.size(), 0);
317 const auto current_batch_size = std::max(target_batch_size, nonempty_quartiles);
318 while (true)
319 {
320 bool has_unseen_rows = false;
321 for (size_t quartile = 0; quartile < quartiles.size(); ++quartile)
322 has_unseen_rows |= positions[quartile] < quartiles[quartile].size();
323 if (!has_unseen_rows)
324 break;
325
326 vector<size_t> counts(quartiles.size(), 0);
327 vector<std::pair<float, size_t>> remainders;
328 const auto allocatable_slots = current_batch_size - nonempty_quartiles;
329 for (size_t quartile = 0; quartile < quartiles.size(); ++quartile) {
330 if (quartiles[quartile].empty())
331 continue;
332 counts[quartile] = 1;
333 const float exact = static_cast<float>(quartiles[quartile].size() * allocatable_slots)
334 / static_cast<float>(total_samples);
335 counts[quartile] += static_cast<size_t>(std::floor(exact));
336 remainders.emplace_back(exact - std::floor(exact), quartile);
337 }
338
339 std::stable_sort(remainders.begin(), remainders.end(),
340 [](const auto& left, const auto& right) {
341 return left.first > right.first;
342 });
343 size_t assigned = 0;
344 for (const auto count : counts)
345 assigned += count;
346 for (size_t i = 0; assigned < current_batch_size; ++i, ++assigned)
347 ++counts[remainders.at(i).second];
348
349 vector<size_t> batch_indices;
350 batch_indices.reserve(current_batch_size);
351 for (size_t quartile = 0; quartile < quartiles.size(); ++quartile) {
352 const auto available = quartiles[quartile].size() - positions[quartile];
353 const auto unique_count = std::min(counts[quartile], available);
354 for (size_t i = 0; i < unique_count; ++i)
355 batch_indices.push_back(quartiles[quartile][positions[quartile]++]);
356 }
357 r.shuffle(batch_indices.begin(), batch_indices.end());
358 batch_cycle_idx.insert(batch_cycle_idx.end(), batch_indices.begin(), batch_indices.end());
359 batch_cycle_sizes.push_back(batch_indices.size());
360 }
361 }
362 }
363
364 // get the indices and move the two cursors
365 const auto count = batch_cycle_sizes.at(batch_cycle_batch_cursor++);
366 vector<size_t> indices(batch_cycle_idx.begin() + batch_cycle_cursor,
367 batch_cycle_idx.begin() + batch_cycle_cursor + count);
368 batch_cycle_cursor += count;
369 return (*this)(indices);
370}

386vector<string> Dataset::get_feature_types() const {
387 // iterate through each feature name in order, get the data type, and return it. This is
388 // used in the python front-end to save the feature types from the training dataset
389 // when calling predict.
390
391 vector<string> python_feature_types;
392 // Iterate through feature_names to preserve order, not through features map
393 for (const auto& name: this->feature_names)
394 {
395 // fmt::print("name:{}\n",name);
396 const auto& value = this->features.at(name);
397
398 // save feature types
399 auto feature_type = StateType(value);
400
401 if (feature_type == DataType::ArrayB)
402 python_feature_types.push_back("ArrayB");
403 else if (feature_type == DataType::ArrayI)
404 python_feature_types.push_back("ArrayI");
405 else if (feature_type == DataType::ArrayF)
406 python_feature_types.push_back("ArrayF");
407 else
408 HANDLE_ERROR_THROW(
409 "get_feature_type does not support the type of this feature yet: " + name +
410 "as a notice, this function is suposed to be used in the python side, to extract data types inferred by Brush type sniffer.");
411 }
412
413 return python_feature_types;
414}

419void Dataset::init()
420{
421 //TODO: populate feature_names, var_data_types, data_types, features_of_type
422 // n_features = this->features.size();
423 // note this will have to change in unsupervised settings
424 // n_samples = this->y.size();
425
426 if (this->features.size() == 0){
427 HANDLE_ERROR_THROW(
428 fmt::format("Error during the initialization of the dataset. It "
429 "does not contain any data\n")
430 );
431 }
432
433 // fmt::print("Dataset::init()\n");
434 // Use the stored original feature name order if available, otherwise iterate through the map.
435 // IMPORTANT: use a value (not a const-ref) to avoid a dangling reference from binding
436 // a const-ref to the temporary produced by a mixed lvalue/prvalue ternary expression.
437 vector<string> names_to_use;
438 if (this->feature_name_order_.empty()) {
439 for (const auto& [name, value] : this->features) {
440 names_to_use.push_back(name);
441 }
442 } else {
443 names_to_use = this->feature_name_order_;
444 }
445
446 for (const auto& name : names_to_use)
447 {
448 // fmt::print("name:{}\n",name);
449 const auto& value = this->features.at(name);
450
451 // save feature types
452 auto feature_type = StateType(value);
453
454 Util::unique_insert(unique_data_types, feature_type);
455 feature_types.push_back( feature_type );
456
457 // add feature to appropriate map list
458 this->features_of_type[feature_type].push_back(name);
459
460 // populate feature names in the original order
461 this->feature_names.push_back(name);
462 }
463
464 // setting the training and validation data indexes
465 auto n_samples = int(this->get_n_samples());
466
467 training_data_idx.resize(0);
468 validation_data_idx.resize(0);
469
470 if (!use_validation)
471 {
472 vector<size_t> idx(n_samples);
473
474 std::iota(idx.begin(), idx.end(), 0);
475
476 std::transform(idx.begin(), idx.end(),
477 back_inserter(training_data_idx),
478 [&](int element) { return element; });
479
480 std::transform(idx.begin(), idx.end(),
481 back_inserter(validation_data_idx),
482 [&](int element) { return element; });
483 }
484 else if (classification) // figuring out training and validation data indexes
485 { // Stratified split for classification problems. TODO: parameters to change stratify behavior? (and set false by default)
486 std::map<float, vector<int>> class_indices; // TODO: I think I can remove many std:: from the code..
487 for (size_t i = 0; i < n_samples; ++i) {
488 class_indices[y[i]].push_back(i);
489 }
490
491 for (auto& class_group : class_indices) {
492 auto& indices = class_group.second;
493
494 int n_class_samples = indices.size();
495
496 vector<size_t> idx(n_class_samples);
497 if (shuffle_split)
498 idx = r.shuffled_index(n_class_samples);
499 else
500 std::iota(idx.begin(), idx.end(), 0);
501
502 auto n_train_samples = int(ceil(n_class_samples*(1.0-validation_size)));
503
504 std::transform(idx.begin(), idx.begin() + n_train_samples,
505 back_inserter(training_data_idx),
506 [&](int element) { return indices[element]; });
507
508 // stratified split so train/validation never overlap when a class is
509 // too small. Now, if a class has no remaining samples for validation,
510 // it contributes only to training (validation gets none for that class).
511 if (n_class_samples - n_train_samples > 0)
512 {
513 std::transform(idx.begin() + n_train_samples, idx.end(),
514 back_inserter(validation_data_idx),
515 [&](int element) { return indices[element]; });
516 }
517 }
518 }
519 else { // regression, or classification without stratification
520 // logic for non-classification problems
521 vector<size_t> idx(n_samples);
522
523 if (shuffle_split)
524 idx = r.shuffled_index(n_samples);
525 else
526 std::iota(idx.begin(), idx.end(), 0);
527
528 // garantee that at least one sample is going to be returned, since
529 // use_batch is true only if batch_size is (0, 1), and ceil will round
530 // up
531 auto n_train_samples = int(ceil(n_samples*(1-validation_size)));
532
533 std::transform(idx.begin(), idx.begin() + n_train_samples,
534 back_inserter(training_data_idx),
535 [&](int element) { return element; });
536
537 if (n_samples - n_train_samples == 0) { // training_data_idx contains all data
538 validation_data_idx = training_data_idx;
539 }
540 else
541 {
542 std::transform(idx.begin() + n_train_samples, idx.end(),
543 back_inserter(validation_data_idx),
544 [&](int element) { return element; });
545 }
546 }
547}

565{
566 // fmt::print("Dataset::make_features()\n");

568

570
571 // check variable names
572 vector<string> tmp_feature_names = {};

574 {

576 for (int i = 0; i < X.cols(); ++i)
577 {

581 }
582 else
583 {
584 if (vn.size() != X.cols())
585 HANDLE_ERROR_THROW(
586 fmt::format("Variable names and data size mismatch: "

589 tmp_feature_names = vn;
590 }
591
592 // check variable types
593 vector<string> var_types;
594 if (ft.empty())
595 {
596 for (int i = 0; i < X.cols(); ++i)
597 {
598 var_types.push_back("");
599 }
600 }
601 else {
602 if (ft.size() != X.cols())
603 HANDLE_ERROR_THROW(
604 fmt::format("Feature type names and data size mismatch: "
605 "{} feature type names and {} features in X",
606 ft.size(), X.cols()) );
607
608 var_types = ft;
609 }
610
611 for (int i = 0; i < X.cols(); ++i)
612 {
613 // fmt::print("X({}): {} \n",i,tmp_feature_names.at(i));
614 State tmp = check_type(X.col(i).array(), var_types.at(i));
615
616 tmp_features[tmp_feature_names.at(i)] = tmp;
617 }
618 // fmt::print("tmp_features insert\n");
619 tmp_features.insert(Z.begin(), Z.end());
620
621 // Store the original feature name order (X features first, then Z features)
622 this->feature_name_order_ = tmp_feature_names;
623 for (const auto& [name, value] : Z) {
624 this->feature_name_order_.push_back(name);
625 }
626
627 return tmp_features;
628};
